      def convolution4(img, kernel, mode = "same"):
        #kernel is 2d array
        K = kernel[::-1, ::-1]
        p = pad(img, kernel.shape, mode)
        H, W = img.shape
        kH, kW = kernel.shape
        H -= kH - 1
        W -= kW - 1
        result = np.zeros((H, W))
        for i in range(H):
            for j in range(W):
                sum = 0
                for h in range(kH):
                    for w in range(kW):
                        sum += p[i + h, j + w] * K[h, w]
                result[i][j] = sum
        return result
